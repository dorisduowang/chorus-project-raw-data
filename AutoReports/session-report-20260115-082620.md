## For External Server (FastAPI)

To receive daily updates, the external server simply needs to:

```python
import subprocess
import json
from pathlib import Path

def pull_latest():
    """Pull latest scan from GitHub."""
    subprocess.run(["git", "pull", "origin", "main"], cwd="/path/to/chorus")

def load_current_state():
    """Load the latest jevans resources."""
    with open("/path/to/chorus/Data/jevans_resources.json") as f:
        return json.load(f)

def load_changes():
    """Load the change log."""
    with open("/path/to/chorus/Data/change_log.json") as f:
        return json.load(f)
```
